cmake_minimum_required(VERSION 3.25)
project(OpenToon VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/utf-8 /permissive-)
endif()
option(OPENTOON_DESKTOP "Build the Qt desktop application" ON)
option(OPENTOON_TESTS "Build behavioral tests" ON)
option(OPENTOON_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(OPENTOON_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
find_package(nlohmann_json 3.12 REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(zstd REQUIRED)
if(NOT TARGET SQLite3::SQLite3)
  add_library(SQLite3::SQLite3 ALIAS SQLite::SQLite3)
endif()
add_library(opentoon_document modules/document/src/document.cpp modules/document/src/timeline.cpp modules/document/src/animation.cpp modules/document/src/key_block.cpp modules/document/src/drawing_selection.cpp)
target_include_directories(opentoon_document PUBLIC modules/document/include)
target_compile_options(opentoon_document PRIVATE $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-Wall;-Wextra;-Wpedantic>)
include(cmake/MyPaint.cmake)
add_library(opentoon_brush adapters/brush/raster_brush.cpp)
target_include_directories(opentoon_brush PUBLIC adapters/brush)
target_link_libraries(opentoon_brush PUBLIC opentoon_document PRIVATE opentoon_mypaint)
add_library(opentoon_application modules/application/src/session.cpp)
target_include_directories(opentoon_application PUBLIC modules/application/include)
target_link_libraries(opentoon_application PUBLIC opentoon_document)
add_library(opentoon_storage adapters/storage/project_store.cpp adapters/storage/serialization.cpp adapters/storage/resources.cpp)
target_include_directories(opentoon_storage PUBLIC adapters/storage)
target_link_libraries(opentoon_storage PUBLIC opentoon_document PRIVATE SQLite3::SQLite3 nlohmann_json::nlohmann_json OpenSSL::Crypto zstd::libzstd_static)
if(OPENTOON_TESTS)
  add_executable(opentoon_brush_benchmark apps/desktop/brush_benchmark.cpp)
  target_link_libraries(opentoon_brush_benchmark PRIVATE opentoon_application opentoon_brush opentoon_storage)
  enable_testing()
  find_package(Catch2 3 REQUIRED)
  add_executable(opentoon_core_tests tests/document_tests.cpp tests/storage_tests.cpp tests/timeline_tests.cpp tests/raster_tests.cpp tests/animation_tests.cpp tests/drawing_selection_tests.cpp)
  target_link_libraries(opentoon_core_tests PRIVATE opentoon_application opentoon_brush opentoon_storage SQLite3::SQLite3 Catch2::Catch2WithMain)
  target_compile_definitions(opentoon_core_tests PRIVATE OPENTOON_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  include(Catch)
  catch_discover_tests(opentoon_core_tests)
endif()
if(OPENTOON_DESKTOP)
  find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Quick QuickControls2 QuickDialogs2 Test)
  qt_standard_project_setup(REQUIRES 6.8)
  add_library(opentoon_render adapters/render/scene_renderer.cpp adapters/render/vector_hit.cpp)
  target_include_directories(opentoon_render PUBLIC adapters/render)
  target_link_libraries(opentoon_render PUBLIC opentoon_document Qt6::Gui)
  qt_add_executable(open-toon MACOSX_BUNDLE apps/desktop/main.cpp apps/desktop/visual_editing_smoke.cpp apps/desktop/key_block_smoke.cpp adapters/qt/editor_controller.cpp adapters/qt/timeline_controller.cpp adapters/qt/animation_controller.cpp adapters/qt/key_selection_controller.cpp adapters/qt/drawing_controller.cpp adapters/qt/editor_controller.h adapters/qt/canvas_item.cpp adapters/qt/canvas_animation.cpp adapters/qt/canvas_cursor.cpp adapters/qt/canvas_item.h)
  target_compile_definitions(open-toon PRIVATE OPENTOON_VERSION="${PROJECT_VERSION}-experimental.7")
  target_include_directories(open-toon PRIVATE adapters/qt)
  target_link_libraries(open-toon PRIVATE opentoon_application opentoon_storage opentoon_brush opentoon_render Qt6::Quick Qt6::QuickControls2 Qt6::QuickDialogs2)
  qt_add_qml_module(open-toon URI OpenToon VERSION 1.0 QML_FILES ui/Main.qml ui/components/ToolButton.qml ui/components/CurveEditor.qml ui/components/MotionOverview.qml ui/components/PoseKeyStrip.qml ui/components/PropertyNumber.qml ui/components/SelectionProperties.qml ui/components/TimingTools.qml ui/components/CompactButton.qml ui/components/CompactComboBox.qml ui/components/CompactSpinBox.qml ui/components/CompactTextField.qml ui/components/CompactCheckBox.qml)
  set_target_properties(open-toon PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "OPEN-TOON" MACOSX_BUNDLE_GUI_IDENTIFIER "org.opentoon.editor" MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}" MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
  install(TARGETS open-toon BUNDLE DESTINATION . RUNTIME DESTINATION bin)
  qt_generate_deploy_qml_app_script(TARGET open-toon OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
  install(SCRIPT ${deploy_script})
  if(OPENTOON_TESTS)
    add_executable(opentoon_render_tests tests/render_tests.cpp)
    target_link_libraries(opentoon_render_tests PRIVATE opentoon_render opentoon_storage Catch2::Catch2WithMain)
    catch_discover_tests(opentoon_render_tests)
    qt_add_executable(opentoon_export_tests tests/export_tests.cpp adapters/qt/editor_controller.cpp adapters/qt/timeline_controller.cpp adapters/qt/animation_controller.cpp adapters/qt/key_selection_controller.cpp adapters/qt/drawing_controller.cpp adapters/qt/editor_controller.h)
    target_include_directories(opentoon_export_tests PRIVATE adapters/qt)
    target_link_libraries(opentoon_export_tests PRIVATE opentoon_application opentoon_storage opentoon_render Catch2::Catch2)
    add_test(NAME export_workflows COMMAND opentoon_export_tests)
    set_tests_properties(export_workflows PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 90)
  endif()
endif()
